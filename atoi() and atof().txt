//  atoi() User defined functions  //

#include<stdio.h>
int my_atoi(char *);
int main()
{
        char s[20];
        int num;
        printf("Enter the String : ");
        scanf("%s",s);

        num=my_atoi(s);

        printf("%d\n",num);
}
int my_atoi(char *p)
{
        int i,num=0,sign=1;

        if (p[0]=='-' || p[0]=='+')
        {
                if (p[0]=='-')
                        sign=-1;
                p++;
        }

        for (i=0;p[i];i++)
        {
                if (p[i]>='0' && p[i]<='9')
                        num=num*10+(p[i]-'0');
                else
                        break;
        }
        return num*sign;
}
